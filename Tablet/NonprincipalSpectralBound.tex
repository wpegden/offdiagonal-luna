\begin{definition}
For a finite loop-aware graph $G$ and $\lambda\in\mathbb R$,
$\operatorname{NonprincipalSpectralBound}(G,\lambda)$ means that every
nonzero real function $v$ on $V(G)$ whose coordinate sum is zero, and every
$\mu\in\mathbb R$ satisfying the adjacency eigen-equations
\[
 \sum_{y\in V(G)}[xy\in E(G)]v(y)=\mu v(x)
 \qquad(x\in V(G)),
\]
has $|\mu|\leq\lambda$.  A loop contributes its diagonal term once.
\end{definition}
